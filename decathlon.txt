Q100. Calculate month-over-month sales growth

with x as (select date_trunc('month',order_date)::date as sales_month ,
       sum(sales_amount)as amount
	   
	   from sport 
	   group by date_trunc('month',order_date)),
previous as(
	   select sales_month ,
	          amount ,
			  lag(amount)over(order by sales_month) as previous_monthly_sales 
			  from x)
select * ,
      round((amount - previous_monthly_sales)/previous_monthly_sales*100,2) as mom_growth
	  from previous
	  order by sales_month
